用固定的三个计数器记录班次

tally.c 为三个班次预留计数。check.c 的三个入口分别检查相互独立、计满拒绝、无效班次拒绝。

/* 自编 C 示例（非原文）：三组检查；每组在新的进程中运行。 */
#include "tally.h"

int main(int argc, char **argv)
{
    if (argc != 2) {
        return 2;
    }
    switch (argv[1][0]) {
    case 'a': /* First and last shifts have separate counters. */
        return !((tally_record(0U) == TALLY_OK) &&
                 (tally_record(2U) == TALLY_OK) &&
                 (tally_read(0U) == 1) && (tally_read(1U) == 0) &&
                 (tally_read(2U) == 1));
    case 'b': /* A full counter does not grow or affect another shift. */
        return !((tally_record(1U) == TALLY_OK) &&
                 (tally_record(1U) == TALLY_OK) &&
                 (tally_record(1U) == TALLY_FULL) &&
                 (tally_read(1U) == 2) && (tally_read(0U) == 0) &&
                 (tally_read(2U) == 0));
    case 'c': /* An unknown shift has no storage assigned to it. */
        return !((tally_record(3U) == TALLY_BAD_SHIFT) &&
                 (tally_read(3U) == -1) && (tally_read(0U) == 0) &&
                 (tally_read(1U) == 0) && (tally_read(2U) == 0));
    default:
        return 2;
    }
}

/* 自编 C 示例（非原文）：三个计数器在整个运行期都存在。 */
#include "tally.h"

#define SHIFT_COUNT 3U
#define SHIFT_LIMIT 2U

static unsigned char totals[SHIFT_COUNT];

enum tally_status tally_record(unsigned int shift)
{
    if (shift >= SHIFT_COUNT) {
        return TALLY_BAD_SHIFT;
    }
    if (totals[shift] >= SHIFT_LIMIT) {
        return TALLY_FULL;
    }
    ++totals[shift];
    return TALLY_OK;
}

int tally_read(unsigned int shift)
{
    return (shift < SHIFT_COUNT) ? (int)totals[shift] : -1;
}

怎么验证，观察到了什么？

用 a、b、c 各启动一个新进程，避免上一项留下状态。三次均返回 0；再看 tally.c，计数的存储来自固定数组。
