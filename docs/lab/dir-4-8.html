不让读表程序依赖计数器的内部布局

表可以存已填格数，也可以存剩余格数；reader.c 只看到一个结构指针和三个操作，而不知道结构成员。

/* 自编 C 示例（非原文） | Only the provider knows the representation. */
#include "meter.h"

struct meter {
#if defined(STORE_REMAINING)
    unsigned char remaining;
#else
    unsigned int filled;
#endif
};

struct meter *meter_reset(void)
{
    static struct meter display;
#if defined(STORE_REMAINING)
    display.remaining = 3U;
#else
    display.filled = 0U;
#endif
    return &display;
}

void meter_mark(struct meter *m)
{
#if defined(STORE_REMAINING)
    if (m->remaining != 0U) { --m->remaining; }
#else
    if (m->filled < 3U) { ++m->filled; }
#endif
}

unsigned int meter_count(const struct meter *m)
{
#if defined(STORE_REMAINING)
    return 3U - m->remaining;
#else
    return m->filled;
#endif
}

/* 自编 C 示例（非原文） | Client: observe the meter only through its API. */
#include <stdio.h>
#include "meter.h"

int main(void)
{
    const unsigned int expected[] = { 0U, 1U, 2U, 3U, 3U };
    struct meter *handle = meter_reset();
    unsigned int i;
    for (i = 0U; i < 5U; ++i) {
        unsigned int visible = meter_count(handle);
        if (visible != expected[i]) { return 1; }
        if (printf("marks=%u\n", visible) < 0) { return 2; }
        if (i < 4U) { meter_mark(handle); }
    }
    return 0;
}

怎么验证，观察到了什么？
